'use strict';
const fs=require('node:fs'),path=require('node:path'),crypto=require('node:crypto'),assert=require('node:assert/strict'),{spawn,spawnSync}=require('node:child_process');
const root=path.resolve(__dirname,'../../..'),recordFile=path.join(root,'plan/acceptance/acceptance.json');
const python=process.env.ACCEPTANCE_PYTHON||'D:/xiang/CX_application/Anconda3/python.exe',routeFile='.agents/skills/acceptance/scripts/route.py';
const read=file=>JSON.parse(fs.readFileSync(file,'utf8')),relative=file=>path.relative(root,file).replaceAll('\\','/');
const ref=file=>({path:relative(file),sha256:crypto.createHash('sha256').update(fs.readFileSync(file)).digest('hex')});
const write=(file,value)=>{fs.mkdirSync(path.dirname(file),{recursive:true});fs.writeFileSync(file,JSON.stringify(value,null,2)+'\n',{flag:'wx'});};
function planner(command,extra=[]){const r=spawnSync(python,['-X','utf8','-B','.agents/skills/acceptance/scripts/execution_plan.py',command,'--workspace','.',...extra],{cwd:root,encoding:'utf8',windowsHide:true,maxBuffer:8*1024*1024});let value;try{value=JSON.parse(r.stdout);}catch{throw Error('Planner returned no valid JSON: '+r.stderr+r.stdout);}if(r.status!==0)throw Error('Planner '+command+' refused: '+JSON.stringify(value));return value;}
function refreshPlan(){planner('sync');planner('check');return read(path.join(root,'plan/acceptance/execution-plan.json'));}
function gate(command,extra=[]){const r=spawnSync(python,['-X','utf8','-B',routeFile,command,'plan/acceptance/acceptance.json','--workspace','.',...extra],{cwd:root,encoding:'utf8',windowsHide:true,maxBuffer:4*1024*1024});let value;try{value=JSON.parse(r.stdout);}catch{throw Error('Gate returned no valid JSON: '+r.stderr+r.stdout);}if(r.status!==0)throw Error('Gate '+command+' refused: '+JSON.stringify(value));if(['begin','submit','close-session'].includes(command))refreshPlan();return value;}
function filesBelow(directory){return fs.readdirSync(directory,{withFileTypes:true}).flatMap(e=>e.isDirectory()?filesBelow(path.join(directory,e.name)):[path.join(directory,e.name)]);}
function frameOf(model){return {documentId:model.document.id,scene:structuredClone(model.scene),inputs:structuredClone(model.inputs),activeIdentity:structuredClone(model.activeIdentity),trace:structuredClone(model.trace),context:structuredClone(model.context),pendingResponse:model.pendingResponse};}
function evidence(folder,definition){
 const files=filesBelow(folder),items=[];
 for(const required of definition.evidence){let selected;
  if(required.form==='assertion-report')continue;
  if(required.form==='screenshot')selected=files.filter(f=>f.endsWith('.png'));
  else if(required.form==='native-tree')selected=files.filter(f=>f.endsWith('.xml'));
  else if(required.form==='layout-measurement')selected=files.filter(f=>/native-frame|responsive-matrix|keyboard/.test(path.basename(f))&&f.endsWith('.json'));
  else if(required.form==='call-trace')selected=files.filter(f=>/trace|observations|summary/.test(path.basename(f))&&f.endsWith('.json'));
  else selected=files.filter(f=>/runtime-frame|contract-tests|task-assertions|page-coverage|inputs|summary/.test(path.basename(f))&&f.endsWith('.json'));
  if(!selected?.length)throw Error('Missing actual '+required.form+' for '+definition.criterion);
  for(const file of selected)items.push({form:required.form,...ref(file),description:required.description});
 }
 return items;
}
async function worker(){
 const {BrowseAdapter}=require('./browse-adapter.cjs'),oracle=require('./browse-oracle.cjs'),supplementary=require('./supplementary.cjs');
 const {UIBrowseModel}=require(path.join(root,'plan/acceptance/.work/runtime/src/verification/ui-browse-model.js'));
 const record=read(recordFile),model=read(path.join(root,record.testCasesPath)),plan=read(path.join(root,record.taskPlanPath)),implementation=read(path.join(root,record.implementationPath));
 const graphFile=path.resolve(path.dirname(path.join(root,record.testCasesPath)),model.stateGraph.path),graph=read(graphFile);
 const seed=read(path.join(root,'plan/acceptance/evidence/preparation/environment/inputs.json'));
 const session=record.sessions.find(s=>s.id===process.env.ACCEPTANCE_SESSION_ID),support=filesBelow(__dirname).filter(f=>f.endsWith('.cjs'));
 assert.ok(session,'A reviewed session must be selected by the supervisor');
 let runtime=new UIBrowseModel(structuredClone(seed)),previous=null;
 const roundControl=require('./round-control.cjs');
 let selectedRound=process.env.ACCEPTANCE_ROUND?roundControl.select(refreshPlan(),process.env.ACCEPTANCE_ROUND,gate('next').nextVisitIndex):null;
 const hostDirectory=path.join(root,'plan/acceptance/evidence/sessions',session.id,'host-'+Date.now());fs.mkdirSync(hostDirectory,{recursive:true});
 const adapter=new BrowseAdapter(hostDirectory);let identity;

  supplementary.baseline(adapter);identity=await adapter.open();await adapter.restart(seed);
  const visual=require('./native-visual.cjs')(root);
  for(;;){
   const next=gate('next'),index=next.nextVisitIndex;if(index>session.to)break;
   if(selectedRound&&index>selectedRound.toVisitIndex){selectedRound=await roundControl.pause(selectedRound,refreshPlan);continue;}
   const planned=planner('next',['--json']);assert.ok(planned.canBegin&&planned.visit?.visitIndex===index,'The live execution plan must authorize this same visit');
   if(index<session.from)throw Error('Session cannot skip earlier required visits');
   const c=model.cases.find(c=>c.id===model.coverage.sequence[index]);assert.ok(c);
   if(c.kind==='primary'&&(!previous||previous.expected.stateId!==c.given.stateId)){
    assert.ok(graph.states.some(s=>s.id===c.given.stateId&&s.isInitial),'Resume from a noninitial state requires reviewed replay evidence');
   }
   const authorization=gate('begin',['--index',String(index)]),authorized=read(recordFile).active;
   const startedAt=new Date().toISOString(),folder=path.join(root,'plan/acceptance/evidence',String(index).padStart(3,'0')+'-'+crypto.randomUUID());fs.mkdirSync(folder,{recursive:true});adapter.directory=folder;adapter.device.directory=folder;
   const checks=[],assertions=[];let givenObserved='Actual initial state has not been observed.',operationObserved='No product operation performed.',targetObserved='No target observation.',blockage=null,observedMocks=[];
   const entry=c.kind==='primary'&&previous&&previous.expected.stateId===c.given.stateId?{kind:'handoff',previousAttemptId:authorized.predecessorAttemptId}:{kind:'initial'};
   try{
    if(c.kind==='primary'){
     if(entry.kind==='initial'){runtime=new UIBrowseModel(structuredClone(seed));await supplementary.ensureBrowse(adapter,seed);await adapter.restart(seed);}
     const beforeRuntime=frameOf(runtime),beforeNative=(await adapter.observe('given-state')).report;
     oracle.matchesState(graph,c.given.stateId,beforeRuntime);oracle.matchesState(graph,c.given.stateId,beforeNative);
     const givenSource=graph.states.find(s=>s.id===c.given.stateId).source.find(s=>s.kind==='ui'),givenDocument=read(path.resolve(path.dirname(graphFile),givenSource.structurePath));require('./binding-oracle.cjs').inspect(givenDocument,beforeRuntime,seed);require('./binding-oracle.cjs').inspect(givenDocument,beforeNative,seed);
     assert.deepEqual(beforeNative.inputs,beforeRuntime.inputs,'Both environments receive the same prepared input');givenObserved='Observed '+c.given.stateId+' in the real native host and the actual Node controller; input objects match.';
     const response=oracle.mockResponse(c);if(response)runtime.queueResponse(structuredClone(response));
     runtime.dispatch(c.operation.input.controlNodeId,c.operation.input.selection);await adapter.perform(c);operationObserved='Dispatched the declared Node event and clicked the actual native control '+c.operation.input.controlNodeId+'.';
     const source=graph.states.find(s=>s.id===c.expected.stateId).source.find(s=>s.kind==='ui'),document=read(path.resolve(path.dirname(graphFile),source.structurePath));await adapter.device.waitSource(x=>x.includes(document.tree.id));
     const actualRuntime=frameOf(runtime),shot=await adapter.observe('target-state'),actualNative=shot.report;
     observedMocks=[{environment:'runtime',calls:actualRuntime.trace.slice(beforeRuntime.trace.length).filter(t=>t.name==='record-response')},{environment:'native',calls:actualNative.trace.slice(beforeNative.trace.length).filter(t=>t.name==='record-response')}];
     write(path.join(folder,'runtime-frame.json'),{before:beforeRuntime,after:actualRuntime});write(path.join(folder,'native-frame.json'),actualNative);
     const captureCheck=(environment,aspect,checkIndex,definition)=>{
      const target='case:'+c.id+':'+aspect+':'+checkIndex+':'+index;let expected,actual,status='passed';
      try{if(environment==='runtime'){oracle.matchesState(graph,c.expected.stateId,actualRuntime);oracle.handoff(c,beforeRuntime,actualRuntime);oracle.responseAssertion(c,beforeRuntime,actualRuntime);require('./binding-oracle.cjs').inspect(document,actualRuntime,seed);expected={from:c.given.stateId,to:c.expected.stateId,control:c.operation.input.controlNodeId,bindingHandoff:true,responseConsumed:true};actual={from:c.given.stateId,to:c.expected.stateId,control:actualRuntime.trace.slice(beforeRuntime.trace.length).find(t=>t.name==='control')?.input.id,bindingHandoff:true,responseConsumed:true};assert.deepEqual(actual,expected);}
       else{oracle.matchesState(graph,c.expected.stateId,actualNative);oracle.handoff(c,beforeNative,actualNative);oracle.responseAssertion(c,beforeNative,actualNative);assert.deepEqual(actualNative.inputs,actualRuntime.inputs);const inspected=visual.inspect(document,actualNative,{xml:shot.evidence.source});expected={documentId:document.id,differences:[],boundaryCalls:0,fontsLoaded:true};actual={documentId:actualNative.documentId,differences:inspected.differences,boundaryCalls:actualNative.boundaryCalls.length,fontsLoaded:Object.values(actualNative.fonts).every(Boolean)};assert.deepEqual(actual,expected);write(path.join(folder,'native-comparisons.json'),inspected);}
      }catch(error){status='failed';expected??={criterion:definition.criterion,oracle:definition.oracle};actual={error:error.stack,observation:environment==='runtime'?actualRuntime:actualNative};}
      assertions.push({target,expected,actual,verdict:status});checks.push({aspect,checkIndex,status,actual:status==='passed'?'Compared the observed '+environment+' result against the independent current specification.':String(actual.error),evidence:evidence(folder,definition)});
     };
     for(const aspect of ['visual','functional'])c.acceptance[aspect].checks.forEach((d,i)=>captureCheck(d.environment,aspect,i,d));targetObserved='Observed native '+actualNative.documentId+' and Node '+actualRuntime.documentId+' after the declared action.';
    }else{
     givenObserved='Prepared the independent fixture described by '+c.given.setup;operationObserved=c.operation.label;

     for(const aspect of ['visual','functional'])for(let i=0;i<c.acceptance[aspect].checks.length;i++){const definition=c.acceptance[aspect].checks[i],result=results[aspect]?.[i];assert.ok(result,'Every declared check needs its actual comparison');const status=require('node:util').isDeepStrictEqual(result.expected,result.actual)?'passed':'failed';assertions.push({target:'case:'+c.id+':'+aspect+':'+i+':'+index,...result,verdict:status});checks.push({aspect,checkIndex:i,status,actual:'Executed the declared isolated fixture and retained its independent comparisons.',evidence:evidence(folder,definition)});}
     targetObserved='Recorded results for the actual independent fixture '+c.id+'.';
    }
   }catch(error){
    try{await adapter.capture('failure-state');}catch{}
    assertions.push({target:'case:'+c.id+':entry:'+index,expected:{given:c.given,operation:c.operation,checks:'all applicable checks observed'},actual:{error:error.stack,givenObserved,operationObserved},verdict:'blocked'});
    blockage={reason:error.message,evidence:[]};checks.length=0;targetObserved='Blocked; inspect the actual error and retained scene evidence.';
   }
   const reportFile=path.join(folder,'assertions.json');write(reportFile,{assertions});
   const reportRef={form:'assertion-report',...ref(reportFile),description:'Independent expected/actual comparisons for this visit.'};for(const check of checks)check.evidence.unshift(reportRef);if(blockage)blockage.evidence=[reportRef,...filesBelow(folder).filter(f=>f.endsWith('.png')).map(f=>({form:'screenshot',...ref(f),description:'Actual scene when preparation or observation was blocked.'}))];
   const contextFile=path.join(folder,'context.json'),environments=[...new Set(checks.length?checks.map(check=>c.acceptance[check.aspect].checks[check.checkIndex].environment):['runtime','native'])];
   const readinessFile=path.join(root,'plan/acceptance/evidence/preparation/environment/readiness.json');
   const prerequisiteObservations={'READY-INTER':'Observed all declared font instances loaded in the real native host; retained the host readiness and current visit comparisons.','READY-UI-HARNESS':'Executed the Node controller and real native UI through the declared driver and recorded this visit observations and independent checks.','READY-NATIVE-HOST':'Connected the actual isolated development package on '+identity.serial+' and recorded its current observations.'};
   const readyPrerequisites=blockage?[]:plan.prerequisites.filter(p=>p.status==='missing'&&p.affects.includes(c.id)).map(p=>{assert.ok(prerequisiteObservations[p.id],'Unsupported preparation prerequisite '+p.id);return {id:p.id,observation:prerequisiteObservations[p.id],evidence:[ref(readinessFile),ref(reportFile)]};});
   write(contextFile,{build:read(path.join(root,'plan/acceptance/evidence/preparation/environment/bundle.json')),inputs:seed,mocks:c.mocks.map(m=>({boundary:m.boundary,trigger:m.trigger,lifetime:m.lifetime,actualCalls:observedMocks.map(o=>({...o,hitCount:o.calls.length})),fixtureTraces:filesBelow(folder).filter(f=>/observations|call-trace|summary/.test(path.basename(f))&&f.endsWith('.json')).map(ref),restored:!blockage&&(c.kind!=='primary'||!runtime.pendingResponse)})),entry,environments:environments.map(environment=>environment==='native'?{environment,host:identity.serial,identity}:{environment,host:process.version}),readyPrerequisites,preparationEvidence:ref(readinessFile),executionScope:{roundId:selectedRound?.id??null,sessionId:session.id}});
   const taskIds=plan.tasks.filter(t=>t.testCaseIds.includes(c.id)).map(t=>t.id),snapshot=[...new Set([...taskIds.flatMap(id=>implementation.tasks[id].files.map(f=>path.join(root,f.path))),...support])].map(ref);
   const run={id:path.basename(folder),taskIds,script:ref(__filename),authoredAt:fs.statSync(__filename).mtime.toISOString(),command:session.command,cwd:session.cwd,startedAt,finishedAt:new Date().toISOString(),exitCode:null,environment:environments.length>1?'mixed':environments[0],files:snapshot,context:ref(contextFile),visits:[{visitIndex:index,entry:entry.kind,givenObserved,operationObserved,targetObserved,outcome:blockage?'blocked':checks.every(c=>c.status==='passed')?'passed':'failed',checks,...(blockage?{blockage}:{})}],taskResults:[]};
   const runFile=path.join(folder,'run.json');write(runFile,run);const submitted=gate('submit',['--token',authorization.token,'--run',relative(runFile)]);console.log(JSON.stringify({index,caseId:c.id,...submitted}));

  }

}
async function supervise(){
 const record=read(recordFile),next=gate('next').nextVisitIndex,session=record.sessions.find(s=>s.from<=next&&next<=s.to);if(!session)throw Error('No unfinished reviewed session');
 const folder=path.join(root,'plan/acceptance/evidence/sessions',session.id,'process-'+Date.now());fs.mkdirSync(folder,{recursive:true});const output=path.join(folder,'output.log'),stream=fs.createWriteStream(output,{flags:'wx'}),startedAt=new Date().toISOString();
 const child=spawn(process.execPath,[__filename,'--worker'],{cwd:root,env:{...process.env,ACCEPTANCE_SESSION_ID:session.id},windowsHide:true,stdio:['ignore','pipe','pipe']});child.stdout.pipe(stream,{end:false});child.stderr.pipe(stream,{end:false});child.stdout.pipe(process.stdout);child.stderr.pipe(process.stderr);
 const exitCode=await new Promise((resolve,reject)=>{child.once('error',reject);child.once('exit',code=>resolve(code??1));});await new Promise(resolve=>stream.end(resolve));const resultFile=path.join(folder,'session-result.json');write(resultFile,{sessionId:session.id,command:session.command,cwd:session.cwd,startedAt,finishedAt:new Date().toISOString(),exitCode,log:ref(output)});
 if(read(recordFile).active===null)gate('close-session',['--run',relative(resultFile)]);process.exitCode=exitCode;
}
if(process.argv.includes('--resume-round'))require('./round-control.cjs').resume(process.argv[process.argv.indexOf('--resume-round')+1]);
else if(process.argv.includes('--inspect'))console.log(JSON.stringify({entry:relative(__filename),environments:['runtime','native'],primary:true,supplementary:['TC-G-EVIDENCE','TC-G-LAYOUT','TC-G-SCOPE','TC-G-VISUAL','TC-REG-CONTAINERS','TC-REG-CONTROLS'],gateProtocol:['plan-next','next','begin','sync-check-plan','observe-and-compare','submit','await-canAdvance','sync-check-plan','pause-at-round-boundary','close-session'],roundSelection:'ACCEPTANCE_ROUND',resumeCommand:'--resume-round <next-round>',productVisitsExecuted:0}));
else (process.argv.includes('--worker')?worker():supervise()).catch(error=>{console.error(error.stack);process.exitCode=1;});
