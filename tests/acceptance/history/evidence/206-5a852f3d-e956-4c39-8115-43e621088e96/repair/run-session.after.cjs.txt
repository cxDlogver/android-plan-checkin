function planner(command,extra=[]){const r=spawnSync(python,['-X','utf8','-B','.agents/skills/acceptance/scripts/execution_plan.py',command,'--workspace','.',...extra],{cwd:root,encoding:'utf8',windowsHide:true,maxBuffer:8*1024*1024});let value;try{value=JSON.parse(r.stdout);}catch{throw Error('Planner returned no valid JSON: '+r.stderr+r.stdout);}if(r.status!==0)throw Error('Planner '+command+' refused: '+JSON.stringify(value));return value;}

function gate(command,extra=[]){
 for(let attempt=0;attempt<16;attempt++){
  const r=spawnSync(python,['-X','utf8','-B',routeFile,command,'plan/acceptance/acceptance.json','--workspace','.',...extra],{cwd:root,encoding:'utf8',windowsHide:true,maxBuffer:4*1024*1024});let value;
  try{value=JSON.parse(r.stdout);}catch{throw Error('Gate returned no valid JSON: '+r.stderr+r.stdout);}
  if(r.status!==0&&value.error?.includes('File exists')&&value.error.includes('acceptance.lock')&&attempt<15){Atomics.wait(new Int32Array(new SharedArrayBuffer(4)),0,0,1000);continue;}
  if(r.status!==0&&!(command==='close-session'&&value.sessionId&&value.status==='failed')&&!(command==='submit'&&value.recorded&&value.synced&&value.checked))throw Error('Gate '+command+' refused: '+JSON.stringify(value));
  if(['begin','submit','close-session'].includes(command))refreshPlan();return value;
 }
 throw Error('The canonical writer did not release its lock');
}
function filesBelow(directory){return fs.readdirSync(directory,{withFileTypes:true}).flatMap(e=>e.isDirectory()?filesBelow(path.join(directory,e.name)):[path.join(directory,e.name)]);}
function frameOf(model){return {documentId:model.document.id,scene:structuredClone(model.scene),inputs:structuredClone(model.inputs),activeIdentity:structuredClone(model.activeIdentity),trace:structuredClone(model.trace),context:structuredClone(model.context),pendingResponse:model.pendingResponse};}
function evidence(folder,definition){
 const files=filesBelow(folder),items=[];
 for(const required of definition.evidence){let selected;
  if(required.form==='assertion-report')continue;
  if(required.form==='screenshot')selected=files.filter(f=>f.endsWith('.png'));
  else if(required.form==='native-tree')selected=files.filter(f=>f.endsWith('.xml'));
  else if(required.form==='layout-measurement')selected=files.filter(f=>/native-frame|responsive-matrix|keyboard/.test(path.basename(f))&&f.endsWith('.json'));
  else if(required.form==='call-trace')selected=files.filter(f=>/trace|observations|summary/.test(path.basename(f))&&f.endsWith('.json'));
  else selected=files.filter(f=>/runtime-frame|contract-tests|task-assertions|page-coverage|inputs|summary/.test(path.basename(f))&&f.endsWith('.json'));
  if(!selected?.length)throw Error('Missing actual '+required.form+' for '+definition.criterion);
  for(const file of selected)items.push({form:required.form,...ref(file),description:required.description});
 }
 return items;
}
function exportTrace(folder,beforeRuntime,actualRuntime,beforeNative,actualNative){

  responseRestored:{runtime:!actualRuntime.pendingResponse,native:!actualNative.pendingResponse}});
 return file;
}
function settledFixtureSeeking(adapter){
 const device=adapter.device,original=device.seek,observations=[];
 device.seek=async function(testID,...args){
  const startedAt=new Date().toISOString();await original.call(this,testID,...args);let last=null,stable=0;const samples=[];
  for(let attempt=0;attempt<40;attempt++){
   const source=await this.source(),tag=source.split('>').find(t=>t.includes('resource-id="'+testID+'"'))||'',match=tag.match(/bounds="\[(-?\d+),(-?\d+)\]\[(-?\d+),(-?\d+)\]"/),bounds=match?.slice(1).map(Number);
   const visible=bounds&&bounds[2]>bounds[0]&&bounds[3]>bounds[1]&&!tag.includes('displayed="false"')&&!tag.includes('enabled="false"'),key=visible?JSON.stringify(bounds):null;
   stable=key&&key===last?stable+1:key?1:0;last=key;samples.push({at:new Date().toISOString(),bounds:bounds??null,visible:Boolean(visible),stable});
   if(stable>=4){const element=await this.id(testID);observations.push({testID,startedAt,samples,element});return element;}
   await new Promise(resolve=>setTimeout(resolve,150));
  }
  observations.push({testID,startedAt,samples,blocked:true});throw Error('Fixture control did not settle at visible native bounds: '+testID);
 };
 return()=>{device.seek=original;write(path.join(adapter.directory,'responsive-matrix-navigation.json'),{minimumConsecutiveStableVisibleSamples:4,sampleIntervalMs:150,operation:'Readiness only; original native click and independent product assertions remain in force.',observations});};
}
function requiredScopeRecord(record,model){

}

async function supervise(){

}

else if(process.argv.includes('--inspect'))console.log(JSON.stringify({entry:relative(__filename),environments:['runtime','native'],primary:true,supplementary:['TC-G-EVIDENCE','TC-G-LAYOUT','TC-G-SCOPE','TC-G-VISUAL','TC-REG-CONTAINERS','TC-REG-CONTROLS'],gateProtocol:['plan-next','next','begin','sync-check-plan','observe-and-compare','submit','await-canAdvance','sync-check-plan','pause-at-round-boundary','close-session'],roundSelection:'ACCEPTANCE_ROUND',resumeCommand:'--resume-round <next-round>',productVisitsExecuted:0}));
