'use strict';
const fs=require('node:fs'),path=require('node:path'),crypto=require('node:crypto'),{spawnSync,execFileSync}=require('node:child_process');
const root=path.resolve(__dirname,'../../..'),directory=path.join(root,'plan/acceptance/evidence/preparation');
const python=process.env.ACCEPTANCE_PYTHON||'D:/xiang/CX_application/Anconda3/python.exe';
fs.mkdirSync(directory,{recursive:true});
const hash=file=>crypto.createHash('sha256').update(fs.readFileSync(file)).digest('hex');
const existing=path.join(root,'plan/acceptance/acceptance.json');if(fs.existsSync(existing)){const record=JSON.parse(fs.readFileSync(existing,'utf8'));if(record.active||record.attempts.length)throw Error('Actual acceptance has started; prepare tooling only through the repair workflow.');}
function run(label,executable,args){const startedAt=new Date().toISOString(),r=spawnSync(executable,args,{cwd:root,encoding:'utf8',windowsHide:true,maxBuffer:16*1024*1024});const result={command:[executable,...args],cwd:root,startedAt,finishedAt:new Date().toISOString(),exitCode:r.status,stdout:r.stdout,stderr:r.stderr,error:r.error?.message};fs.writeFileSync(path.join(directory,label+'.json'),JSON.stringify(result,null,2)+'\n');if(r.status!==0)throw Error(label+': '+(r.stderr||r.stdout||r.error?.message));return result;}
const originals=['plan/task-planning/test-cases.json','plan/task-planning/test-cases.html','plan/task-planning/task-plan.json','plan/task-planning/task-plan.md'];
const baseline=path.join(directory,'planning-originals.json');
if(!fs.existsSync(baseline))fs.writeFileSync(baseline,JSON.stringify(originals.map(file=>({path:file,sha256:hash(path.join(root,file))})),null,2)+'\n');
for(const ref of JSON.parse(fs.readFileSync(baseline,'utf8')))if(hash(path.join(root,ref.path))!==ref.sha256)throw Error('Planning original changed: '+ref.path);
const preflight=run('preflight',python,['-X','utf8','-B','.agents/skills/acceptance/scripts/route.py','preflight','--workspace','.', '--cases','plan/task-planning/test-cases.json','--plan','plan/task-planning/task-plan.json','--implementation','plan/code-execution/implementation.json']);
if(JSON.parse(preflight.stdout).status!=='ready')throw Error('Preflight did not authorize acceptance preparation');
const revision=execFileSync('git',['rev-parse','HEAD'],{cwd:root,encoding:'utf8',windowsHide:true}).trim();
const modules=['browse-adapter.cjs','browse-oracle.cjs','native-visual.cjs','browse-matrix.cjs','native-fixtures.cjs','accept-p02.cjs','accept-p03.cjs','accept-p04.cjs','accept-p05.cjs','accept-p06.cjs'];
const provenance=[];
for(const name of modules){const sourcePath='plan/code-execution/scripts/'+name;let source=execFileSync('git',['show',revision+':'+sourcePath],{cwd:root,encoding:'utf8',windowsHide:true,maxBuffer:4*1024*1024});const originalSha256=crypto.createHash('sha256').update(source).digest('hex');source=source.replaceAll('.cache/ui-tests','plan/acceptance/.work/runtime');const file=path.join(__dirname,name);if(!fs.existsSync(file))fs.writeFileSync(file,source);provenance.push({path:path.relative(root,file).replaceAll('\\','/'),sourceRevision:revision,sourcePath,sourceSha256:originalSha256,currentSha256:hash(file),adaptations:['Relocate compiled runtime imports to the acceptance artifact root; no historical test results are reused.']});}
fs.writeFileSync(path.join(directory,'driver-provenance.json'),JSON.stringify(provenance,null,2)+'\n');
run('runtime-compile',process.execPath,['node_modules/typescript/bin/tsc','-p','tsconfig.ui-tests.json','--outDir','plan/acceptance/.work/runtime']);
const syntax=[];
for(const name of fs.readdirSync(__dirname).filter(n=>n.endsWith('.cjs'))){const r=spawnSync(process.execPath,['--check',path.join(__dirname,name)],{cwd:root,encoding:'utf8',windowsHide:true});syntax.push({path:'plan/acceptance/scripts/'+name,exitCode:r.status,stdout:r.stdout,stderr:r.stderr});if(r.status!==0)throw Error(name+': '+r.stderr);}
fs.writeFileSync(path.join(directory,'driver-syntax.json'),JSON.stringify(syntax,null,2)+'\n');
console.log(JSON.stringify({preflight:'ready',sourceRevision:revision,compiledRuntime:true,syntaxChecked:syntax.length,productVisitsExecuted:0}));
