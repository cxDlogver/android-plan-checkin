const fs=require('node:fs');
const path=require('node:path');
const assert=require('node:assert/strict');
const {execFileSync}=require('node:child_process');
const {env,toolRoot}=require('./task-env.cjs');
const elementKey='element-6066-11e4-a52e-4f735466cecf';
const decode=text=>text.replace(/&quot;/g,'"').replace(/&apos;/g,"'").replace(/&lt;/g,'<').replace(/&gt;/g,'>').replace(/&#(\d+);/g,(_,n)=>String.fromCodePoint(Number(n))).replace(/&amp;/g,'&');
function reportFromSource(source){const tag=source.match(/<[^>]*resource-id="native\.report"[^>]*>/)?.[0];const content=tag?.match(/content-desc=(["'])(.*?)\1/)?.[2];return content&&decode(content)!=='null'?JSON.parse(decode(content)):null;}
class DeviceDriver{
  constructor(api,directory){assert.ok(['29','36'].includes(String(api)));this.api=String(api);this.serial=api==='29'||api===29?'emulator-5560':'emulator-5562';this.directory=directory;this.package='com.xiang.plancheckin.dev';this.address='http://127.0.0.1:4723';fs.mkdirSync(directory,{recursive:true});}
  adb(args){return execFileSync(path.join(toolRoot,'sdk/platform-tools/adb.exe'),['-s',this.serial,...args],{env,encoding:'utf8',timeout:180000,windowsHide:true});}
  identity(){const avd=this.adb(['emu','avd','name']).split(/\r?\n/).map(line=>line.trim());assert.ok(avd.includes(`plancheckin_api${this.api}`),'Only the project disposable AVD is authorized.');const actualApi=this.adb(['shell','getprop','ro.build.version.sdk']).trim();assert.equal(actualApi,this.api);return {serial:this.serial,avd:`plancheckin_api${this.api}`,api:Number(actualApi),package:this.package,display:this.adb(['shell','wm','size']).trim(),density:this.adb(['shell','wm','density']).trim(),fontScale:this.adb(['shell','settings','get','system','font_scale']).trim(),timezone:this.adb(['shell','getprop','persist.sys.timezone']).trim()};}
  async request(endpoint,method='GET',body){const response=await fetch(this.address+endpoint,{method,headers:{'content-type':'application/json'},...(body?{body:JSON.stringify(body)}:{}),signal:AbortSignal.timeout(180000)});const result=await response.json();if(!response.ok||result.value?.error)throw new Error(`${method} ${endpoint}: ${result.value?.message||response.status}`);return result.value;}
  endpoint(suffix){assert.ok(this.sessionId,'No Appium session');return `/session/${this.sessionId}/${suffix}`;}
  async open(){this.identity();const session=await this.request('/session','POST',{capabilities:{alwaysMatch:{platformName:'Android','appium:automationName':'UiAutomator2','appium:udid':this.serial,'appium:appPackage':this.package,'appium:appActivity':'.MainActivity','appium:noReset':true,'appium:forceAppLaunch':true,'appium:shouldTerminateApp':true,'appium:newCommandTimeout':240,'appium:uiautomator2ServerInstallTimeout':120000,'appium:uiautomator2ServerLaunchTimeout':120000,...(this.capabilities||{})},firstMatch:[{}]}});this.sessionId=session.sessionId;assert.ok(this.sessionId);}
  async close(){if(this.sessionId){await this.request(`/session/${this.sessionId}`,'DELETE');this.sessionId=null;}}
  async source(){return this.request(this.endpoint('source'));}
  async save(name){const source=await this.source();fs.writeFileSync(`${this.directory}/${name}.xml`,source);const screenshot=await this.request(this.endpoint('screenshot'));fs.writeFileSync(`${this.directory}/${name}.png`,Buffer.from(screenshot,'base64'));return source;}
  async element(using,value){const found=await this.request(this.endpoint('element'),'POST',{using,value});assert.ok(found[elementKey]);return found[elementKey];}
  async id(testID){for(let attempt=0;attempt<4;attempt++){try{return await this.element('xpath',`//*[@resource-id="${testID}"]`);}catch(error){if(attempt===3||!/could not be located|no such element|stale element|Cached elements/i.test(error.message))throw error;await new Promise(resolve=>setTimeout(resolve,350));}}}
  async click(testID){const id=await this.id(testID);await this.request(this.endpoint(`element/${id}/click`),'POST',{});}
  async text(text){const id=await this.element('-android uiautomator',`new UiSelector().text(${JSON.stringify(text)})`);await this.request(this.endpoint(`element/${id}/click`),'POST',{});}
  async textVisible(text,maxScrolls=10,percent=.8){for(const direction of ['down','up']){let settledEnd=0;for(let attempt=0;attempt<maxScrolls;attempt++){const source=await this.source();if([...source.matchAll(/\btext="([^"]*)"/g)].some(match=>decode(match[1])===text))return this.text(text);if(await this.scroll(direction,percent)===false){if(++settledEnd===3)break;await new Promise(resolve=>setTimeout(resolve,500));}else settledEnd=0;}}throw new Error(`Product text control unavailable: ${text}`);}
  async seeText(text,timeout=20000){const deadline=Date.now()+timeout;do{for(const direction of ['down','up'])for(let attempt=0;attempt<10&&Date.now()<deadline;attempt++){const source=await this.source();if(source.includes(text))return source;if(await this.scroll(direction)===false)break;}await new Promise(resolve=>setTimeout(resolve,500));}while(Date.now()<deadline);throw new Error(`Product text did not appear: ${text}`);}
  async input(testID,text){let id=await this.id(testID);await this.request(this.endpoint(`element/${id}/click`),'POST',{});await this.request(this.endpoint(`element/${id}/clear`),'POST',{});id=await this.id(testID);await this.request(this.endpoint(`element/${id}/value`),'POST',{text});}
  async seek(testID,outerGutter=true){for(const direction of ['down','up']){let settledEnd=0;for(let attempt=0;attempt<16;attempt++){const source=await this.source();if(source.includes(`resource-id="${testID}"`)){try{return await this.id(testID);}catch(error){if(!/could not be located|no such element|stale element|Cached elements/i.test(error.message))throw error;await this.source();}}if(await this.scroll(direction,.8,outerGutter)===false){if(++settledEnd===3)break;await new Promise(resolve=>setTimeout(resolve,500));}else settledEnd=0;}}const source=await this.source();if(source.includes(`resource-id="${testID}"`))return this.id(testID);throw new Error(`Product control unavailable: ${testID}`);}
  async clickVisible(testID){const id=/^(?:record|plan)\.(?:save|reconfirm)$/.test(testID)?await this.revealButton(testID):await this.seek(testID);await this.request(this.endpoint(`element/${id}/click`),'POST',{});}
  async revealButton(testID,minHeightDp=48){const density=[...this.identity().density.matchAll(/\d+/g)].at(-1)[0],minimum=minHeightDp*Number(density)/160;for(const direction of ['down','up']){let settledEnd=0;for(let i=0;i<20;i++){const xml=await this.source(),node=xml.match(new RegExp('<[^>]*resource-id="'+testID.replace(/[.*+?^${}()|[\]\\]/g,'\\$&')+'"[^>]*>'))?.[0],b=node?.match(/bounds="\[(\d+),(\d+)\]\[(\d+),(\d+)\]"/);if(b&&Number(b[4])-Number(b[2])>=minimum)return this.id(testID);if(await this.scroll(direction,.65,true)===false){if(++settledEnd===3)break;await new Promise(r=>setTimeout(r,500));}else settledEnd=0;}}throw new Error(`Button does not expose a complete ${minHeightDp} dp hit area: ${testID}`);}
  async hideKeyboard(){
    if(this.api!=='36'){await this.request(this.endpoint('appium/device/hide_keyboard'),'POST',{});return;}
    // API 36 omits the legacy mIsInputViewShown field used by Appium's helper.
    // Observe the actual IME state before sending Back so a hidden IME never pops a form.
    const before=this.adb(['shell','dumpsys','input_method']);if(!/mInputShown=true\b/.test(before))return;
    await this.back();const deadline=Date.now()+5000;let after;
    do{after=this.adb(['shell','dumpsys','input_method']);if(/mInputShown=false\b/.test(after)){fs.appendFileSync(path.join(this.directory,'keyboard-dismiss-observations.jsonl'),JSON.stringify({at:new Date().toISOString(),api:36,action:'native KEYCODE_BACK only while mInputShown=true',beforeShown:true,afterShown:false,legacyInputViewFieldPresent:/mIsInputViewShown=/.test(before)})+'\n');return;}await new Promise(r=>setTimeout(r,250));}while(Date.now()<deadline);
    throw new Error('Native API 36 IME remained visible after keyboard dismissal');
  }
  async back(){await this.request(this.endpoint('appium/device/press_keycode'),'POST',{keycode:4});}
  async waitSource(predicate,timeout=60000){const deadline=Date.now()+timeout;let source;do{try{source=await this.source();if(predicate(source))return source;}catch(error){if(!/waiting for the root AccessibilityNodeInfo/.test(error.message))throw error;}await new Promise(resolve=>setTimeout(resolve,750));}while(Date.now()<deadline);await this.save('unexpected');throw new Error('Expected product UI did not appear; inspect unexpected.xml/png.');}
  async scroll(direction='down',percent=.8,outerGutter=true){
    for(let attempt=0;attempt<3;attempt++){
      const elements=await this.request(this.endpoint('elements'),'POST',{using:'xpath',value:`//*[@scrollable="true" and @package="${this.package}" and not(self::android.widget.EditText)]`});
      const origin=elements[0]?.[elementKey];
      try{if(outerGutter&&origin){const rect=await this.request(this.endpoint(`element/${origin}/rect`)),before=await this.source(),x=Math.round(rect.x+Math.min(38,rect.width*.1)),top=Math.round(rect.y+90),bottom=Math.round(rect.y+rect.height-180),distance=Math.round((bottom-top)*Math.min(percent,.9)),start=direction==='down'?bottom:top,end=direction==='down'?bottom-distance:top+distance;await this.request(this.endpoint('actions'),'POST',{actions:[{type:'pointer',id:'outer-scroll-finger',parameters:{pointerType:'touch'},actions:[{type:'pointerMove',duration:0,x,y:start},{type:'pointerDown',button:0},{type:'pointerMove',duration:450,x,y:end},{type:'pointerUp',button:0}]}]});const after=await this.source();if(before!==after)return true;await this.request(this.endpoint('execute/sync'),'POST',{script:'mobile: scrollGesture',args:[{elementId:origin,direction,percent}]});const changed=before!==await this.source();fs.appendFileSync(path.join(this.directory,'scroll-fallback-observations.jsonl'),JSON.stringify({at:new Date().toISOString(),direction,percent,scope:'Native scrollGesture only after the outer-gutter pointer left hierarchy unchanged',changed})+'\n');return changed;}return await this.request(this.endpoint('execute/sync'),'POST',{script:'mobile: scrollGesture',args:[origin?{elementId:origin,direction,percent}:{left:20,top:180,width:1000,height:1750,direction,percent}]});}
      catch(error){if(attempt===2||!(/Cached elements|stale element/i.test(error.message)))throw error;await this.source();}
    }
  }
  async report(){for(let attempt=0;attempt<12;attempt++){const report=reportFromSource(await this.source());if(report)return report;await this.scroll();}throw new Error('Native diagnostics report unavailable.');}
  deepLink(url){this.adb(['shell','am','start','-W','-a','android.intent.action.VIEW','-d',url,this.package]);}
}
module.exports={DeviceDriver,decode,reportFromSource};
